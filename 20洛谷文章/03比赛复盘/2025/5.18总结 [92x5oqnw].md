# 题目总结
## T1
### 错误原因
> 骗分
### 正确代码

```cpp
#include<bits/stdc++.h>
using namespace std;
struct node{
    int x,y;
}a[1010];
bool cmp(node a,node b){
    return a.x<b.x;
}
int n,ans;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i].x>>a[i].y;
    }
    sort(a+1,a+n+1,cmp);
    for(int i=1;i<=n;i++){
        for(int j=i+1;j<=n;j++){
            if(a[i].x<=a[j].y){
                ans=max(ans,min(a[i].y,a[j].y)-a[j].x);
            }
        }
    }
    cout<<ans;
    return 0;
}
```
