## T2
### 错误原因
>写的暴力，时间复杂度超了。

### 错误代码（莫名其妙的80分）
```cpp
##include<bits/stdc++.h>
using namespace std;
long long a[100010];
int n;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n;i++){
        for(int j=i+1;j<=n;j++){
            for(int k=j+1;k<=n;k++){
                if((a[i]+a[j]+a[k])%10==3){
                    puts("YES");
                    return 0;
                }
            }
        }
    }
    puts("NO");
    return 0;
}
```

### 正确思路

优化！！！  
十年OI一场空，不开 $long$ $long$ 见祖宗(开了$long$ $long$ (≧▽≦q))

### 正确代码
```cpp
#include<bits/stdc++.h>
using namespace std;
int d[10];
long long a[200010],n;
int main(){
    cin>>n;
    for(int i=1;i<=n;i++){
        cin>>a[i];
        d[a[i]%10]++;
    }
    for(int i=0;i<=9;i++){
        for(int j=0;j<=9;j++){
            for(int k=0;k<=9;k++){
                if((i+j+k)%10==3){
                    d[i]--;
                    d[j]--;
                    d[k]--;
                    if(d[i]<0||d[j]<0||d[k]<0){
                        d[i]++;
                        d[j]++;
                        d[k]++;
                        continue;
                    }
                    puts("YES");
                    return 0;
                }
            }
        }
    }
    puts("NO");
    return 0;
}
```

```cpp
#include<bits/stdc++.h>
using namespace std;
int a[200010],n,r;
int main(){
    cin>>n;
    while(n--){
        int x;
        cin>>x;
        a[++r]=x;
        if(r<=1) continue;
        else if(a[r]!=a[r-1]) continue;
        else if(a[r]==a[r-1]){
            while(r>=2&&a[r]==a[r-1]){
                int t=a[r]+1;
                a[r-1]=t;
                r--;
            }
        }
    }
    cout<<r;
    return 0;
}
```

## T4
### 错误原因
>写的暴力，时间复杂度超了。

#### 暴力思路：
>优化！！！

#### 暴力代码：
```cpp
#include<bits/stdc++.h>
using namespace std;
int n;
long long a[1010];
int main(){
    cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int x=1;x<=n;x++){
        for(int y=1;y<=n;y++){
            for(int z=1;z<=n;z++){
                for(int l=1;l<=n;l++){
                    if(a[x]+a[y]+a[z]==a[l]){
                        puts("Yes");
                        return 0;
                    }
                }
            }
        }
    }
    puts("No");
    return 0;
}
```

#### 暴力时间复杂度：
$O(n^4)$

### 正确思路
使用map优化。

### 正确代码
```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
int n, m, l, r, k, vis1[1000005], vis2[1000005], vis3[1000005], diffa[1000005], diffb[1000005], diffc[1000005];
signed main()
{
	cin>>n>>m;
	for(int i=1;i<=m;i++)
	{
		cin>>l>>r>>k;
		if(k==1)
		{
			diffa[l]++,diffa[r+1]--;
		}
		if(k==2)
		{
			diffb[l]++,diffb[r+1]--;
		}
		if(k==3)
		{
			diffc[l]++,diffc[r+1]--;
		}
	}
	for(int i=1;i<=n;i++)
	{
		vis1[i]=vis1[i-1]+diffa[i];
		vis2[i]=vis2[i-1]+diffb[i];
		vis3[i]=vis3[i-1]+diffc[i];
	}
	int cnt=0;
	for(int i=1;i<=n;i++)
	{
		if(vis1[i]>0&&vis2[i]>0&&vis3[i]==0)
		{
			cnt++;
		}
	}
	cout<<cnt;
	return 0;
}
```
