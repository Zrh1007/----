## 【深基7.例3】闰年展示
### 错误原因：
审题错误  
### 更正的代码：

```cpp
#include <bits/stdc++.h>
using namespace std;
int x,y,cnt;
int main(){
    cin>>x>>y;
    for(int i=x;i<=y;i++){
    	if((i%4==0&&i%100!=0)||(i%400==0)){
    		cnt++;
		}
	}
	cout<<cnt<<'\n';
	for(int i=x;i<=y;i++){
    	if((i%4==0&&i%100!=0)||(i%400==0)){
    		cout<<i<<' ';
		}
	}
    return 0;
}
```
## [GESP202406 四级] 黑白方块
### 错误原因：
代码细节问题
### 更正的代码：

```cpp
#include <bits/stdc++.h>
using namespace std;
int n,m,ma=0,a[12][12];
int main(){
	cin>>n>>m;
	for(int i=1;i<=n;i++){
		for(int j=1;j<=m;j++){
			char c; cin>>c;
			if(c=='1') a[i][j]=1;
			else a[i][j]=0;
		}
	}
	for(int i=1;i<=n;i++){
		for(int j=1;j<=m;j++){
			for(int k=1;k<=n;k++){
				for(int l=1;l<=m;l++){
					int cnt=0,c0=0,c1=0;
					for(int x=i;x<=k;x++){
						for(int y=j;y<=l;y++){
							cnt++;
							if(a[x][y]==0) c0++;
							else c1++;
						}
					}
					if(c0==c1) ma=max(ma,cnt);
				}
			}
		}
	}
	cout<<ma;
    return 0;
}
```
