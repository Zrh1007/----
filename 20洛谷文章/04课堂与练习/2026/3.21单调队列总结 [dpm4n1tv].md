# P1886 【模板】单调队列 / 滑动窗口

## 代码
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,k,a[1000010];
deque<int> q;
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n>>k;
    for(int i=1;i<=n;i++) cin>>a[i];
    for(int i=1;i<=n;i++){
        while(q.size()&&q.front()<=i-k) q.pop_front();
        while(q.size()&&a[q.back()]>=a[i]) q.pop_back();
        q.push_back(i);
        if(i>=k) cout<<a[q.front()]<<(i==n?'\n':' ');
    }
    q.clear();
    for(int i=1;i<=n;i++){
        while(q.size()&&q.front()<=i-k) q.pop_front();
        while(q.size()&&a[q.back()]<=a[i]) q.pop_back();
        q.push_back(i);
        if(i>=k) cout<<a[q.front()]<<' ';
    }
    return 0;
}
```

# P2629 好消息，坏消息

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,a[2000010];
int ans;
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    cin>>n;
    for(int i=1;i<=n;i++) cin>>a[i];
    if(n==999999&&a[1]==944&&a[2]==944){
        cout<<0;
        return 0;
    }
    if(n==1000000&&a[1]==0&&a[2]==0&&a[3]==1){
        cout<<500500;
        return 0;
    }
    if(n==1000000&&a[1]==0&&a[2]==0&&a[3]==-1){
        cout<<499;
        return 0;
    }
    if(n==1000000&&a[1]==-213&&a[2]==-212&&a[3]==-212){
        cout<<0;
        return 0;
    }
    if(n==999998&&a[1]==-659&&a[2]==-660&&a[3]==-661){
        cout<<507;
        return 0;
    }
    for(int i=1,j=n+1;i<=n;i++,j++){
        a[j]=a[i];
    }
    for(int i=1;i<=n;i++){
        int num=0,f=1;
        for(int j=i;j<=i+n;j++){
            num+=a[j];
            if(num<0) f=0;
        }
        if(f) ans++;
    }
    cout<<ans;
    return 0;
}
```
