# B3666 求数列所有后缀最大值的位置

```cpp
#include<bits/stdc++.h>
using namespace std;
unsigned long long n,a[1000010];
stack<unsigned long long> s;
unsigned long long ans;
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n;
    for(unsigned long long i=1;i<=n;i++) cin>>a[i];
    for(unsigned long long i=1;i<=n;i++){
        while(s.size()&&a[s.top()]<=a[i]){
            ans^=s.top();
            s.pop();
        }
        s.push(i);
        ans^=i;
        cout<<ans<<'\n';
    }
    return 0;
}
```

# P3512 [POI 2010] PIL-Pilots

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=3000005;
long long k,minn[N],maxx[N],x;
int n,minid[N],maxid[N];
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>k>>n;
    int l=1,ans=0,minh=1,mint=0,maxh=1,maxt=0;
    for(int r=1;r<=n;r++){
        cin>>x;
        while(minh<=mint&&minn[mint]>=x) mint--;
        minn[++mint]=x;
        minid[mint]=r;
        while(maxh<=maxt&&maxx[maxt]<=x) maxt--;
        maxx[++maxt]=x;
        maxid[maxt]=r;
        while(maxh<=maxt&&minh<=mint&&maxx[maxh]-minn[minh]>k){
            if(minid[minh]==l) minh++;
            if(maxid[maxh]==l) maxh++;
            l++;
        }
        ans=max(ans,r-l+1);
    }
    cout<<ans<<'\n';
    return 0;
}
```
