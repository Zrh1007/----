# P3374 【模板】树状数组 1

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,opt,x,y;
long long c[500005];
int lowbit(int x){
    return x&-x;
}
void update(int x,int k){
    while(x<=n){
        c[x]+=k;
        x+=lowbit(x);
    }
}
long long query(int x){
    long long sum=0;
    while(x>0){
        sum+=c[x];
        x-=lowbit(x);
    }
    return sum;
}
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        cin>>x;
        update(i,x);
    }
    while(m--){
        cin>>opt>>x>>y;
        if(opt==1) update(x,y);
        else cout<<query(y)-query(x-1)<<'\n';
    }
    return 0;
}
```

# P3368 【模板】树状数组 2

```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,opt,x,y;
long long c[500005];
int lowbit(int x){
    return x&-x;
}
void update(int x,long long k){
    while(x<=n){
        c[x]+=k;
        x+=lowbit(x);
    }
}
long long query(int x){
    long long sum=0;
    while(x>0){
        sum+=c[x];
        x-=lowbit(x);
    }
    return sum;
}
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n>>m;
    long long now,last=0,k;
    for(int i=1;i<=n;i++){
        cin>>now;
        update(i,now-last);
        last=now;
    }
    while(m--){
        cin>>opt>>x;
        if(opt==1){
            cin>>y>>k;
            update(x,k);
            update(y+1,-k);
        }
        else cout<<query(x)<<'\n';
    }
    return 0;
}
```
