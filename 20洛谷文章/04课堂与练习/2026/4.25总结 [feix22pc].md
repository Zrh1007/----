```cpp
#include<bits/stdc++.h>
using namespace std;
long long n,len,tot,a[300010],id[300010],L[1010],R[1010],tag[1010];

    len=sqrt(n);
    for(int i=1;i<=n;i++){
        cin>>a[i];
        id[i]=(i-1)/len+1;
    }
    tot=n/len;
    for(int i=1;i<=tot;i++){
        L[i]=(i-1)*len+1;
        R[i]=len*i;
    }
    if(n%len!=0){
        tot++;
        L[tot]=R[tot-1]+1;
        R[tot]=n;
    }
    return;
}

    long long left=id[l],right=id[r];
    if(left==right){
        for(long long i=l;i<=r;i++) a[i]+=c;
    }
    else{
        for(int i=left+1;i<right;i++) tag[i]+=c;
        for(long long i=l;i<=R[left];i++) a[i]+=c;
        for(long long i=L[right];i<=r;i++) a[i]+=c;
    }
}
```

# P13979 数列分块入门 4
```cpp
#include<bits/stdc++.h>
using namespace std;
long long n,len,tot,a[300010],id[300010],L[1010],R[1010],tag[1010],sum[1010];
void init(){
    len=sqrt(n);
    for(int i=1;i<=n;i++){
        cin>>a[i];
        id[i]=(i-1)/len+1;
    }
    tot=n/len;
    for(int i=1;i<=tot;i++){
        L[i]=(i-1)*len+1;
        R[i]=len*i;
    }
    if(n%len!=0){
        tot++;
        L[tot]=R[tot-1]+1;
        R[tot]=n;
    }
    for(int i=1;i<=n;i++){
        sum[id[i]]+=a[i];
    }
    return;
}
void add(long long l,long long r,long long c){
    long long left=id[l],right=id[r];
    if(left==right){
        for(long long i=l;i<=r;i++){
            a[i]+=c;
            sum[left]+=c;
        }
    }
    else{
        for(int i=left+1;i<right;i++){
            tag[i]+=c;
        }
        for(long long i=l;i<=R[left];i++){
            a[i]+=c;
            sum[left]+=c;
        }
        for(long long i=L[right];i<=r;i++){
            a[i]+=c;
            sum[right]+=c;
        }
    }
}
long long query(long long l,long long r,long long mod){
    long long left=id[l],right=id[r],ans=0;
    if(left==right){
        for(long long i=l;i<=r;i++){
            ans+=a[i]+tag[left];
        }
    }
    else{
        for(long long i=l;i<=R[left];i++){
            ans+=a[i]+tag[left];
        }
        for(int i=left+1;i<right;i++){
            ans+=sum[i]+tag[i]*(R[i]-L[i]+1);
        }
        for(long long i=L[right];i<=r;i++){
            ans+=a[i]+tag[right];
        }
    }
    ans%=mod;
    if(ans<0) ans+=mod;
    return ans;
}
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n;
    init();
    for(long long i=1,op,l,r,c;i<=n;i++){
        cin>>op>>l>>r>>c;
        if(op==0){
            add(l,r,c);
        }
        else{
            cout<<query(l,r,c+1)<<'\n';
        }
    }
    return 0;
}
```
