# P3865 【模板】ST 表 & RMQ 问题
```cpp
#include<bits/stdc++.h>
using namespace std;
int dp[100010][30];
int n,m,a[100010];
void init(){
    memset(dp,0,sizeof(dp));
    for(int i=1;i<=n;i++) dp[i][0]=a[i];
    for(int j=1;j<=log2(n);j++){
        for(int i=1;i+(1<<j)-1<=n;i++){
            dp[i][j]=max(dp[i][j-1],dp[i+(1<<(j-1))][j-1]);
        }
    }
}
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n>>m;
    for(int i=1;i<=n;i++) cin>>a[i];
    init();
    while(m--){
        int l,r;
        cin>>l>>r;
        int k=log2(r-l+1);
        cout<<max(dp[l][k],dp[r-(1<<k)+1][k])<<'\n';
    }
    return 0;
}
```

# T752170 策略游戏
```cpp
#include<bits/stdc++.h>
using namespace std;
typedef long long ll;
ll n,m,q,a[200010],b[200010],sum[200010];
ll dp1[200010][30],dp2[200010][30],dp3[200010][30],dp4[200010][30],dp5[200010][30],dp6[200010][30];
int lg[200010];
void init(){
	memset(dp1,-0x3f,sizeof dp1);
	memset(dp2,0x3f,sizeof dp2);
	memset(dp3,-0x3f,sizeof dp3);
	memset(dp4,0x3f,sizeof dp4);
	memset(dp5,-0x3f,sizeof dp5);
	memset(dp6,0x3f,sizeof dp6);
	lg[0]=-1;
	for(int i=1;i<200010;i++) lg[i]=lg[i>>1]+1;
	for(int i=1;i<=n;i++){
		dp1[i][0]=dp2[i][0]=a[i];
		if(a[i]<0) dp5[i][0]=a[i];
		if(a[i]>0) dp6[i][0]=a[i];
		sum[i]=sum[i-1]+(a[i]==0);
	}
	for(int i=1;i<=m;i++) dp3[i][0]=dp4[i][0]=b[i];
	for(int j=1;j<=lg[n];j++){
		for(int i=1;i+(1<<j)-1<=n;i++){
			dp1[i][j]=max(dp1[i][j-1],dp1[i+(1<<(j-1))][j-1]);
			dp2[i][j]=min(dp2[i][j-1],dp2[i+(1<<(j-1))][j-1]);
			dp5[i][j]=max(dp5[i][j-1],dp5[i+(1<<(j-1))][j-1]);
			dp6[i][j]=min(dp6[i][j-1],dp6[i+(1<<(j-1))][j-1]);
		}
	}
	for(int j=1;j<=lg[m];j++){
		for(int i=1;i+(1<<j)-1<=m;i++){
			dp3[i][j]=max(dp3[i][j-1],dp3[i+(1<<(j-1))][j-1]);
			dp4[i][j]=min(dp4[i][j-1],dp4[i+(1<<(j-1))][j-1]);
		}
	}
	return;
}
ll query_max(ll dp[][30],int l,int r){
	int x=lg[r-l+1];
	return max(dp[l][x],dp[r-(1<<x)+1][x]);
}
ll query_min(ll dp[][30],int l,int r){
	int x=lg[r-l+1];
	return min(dp[l][x],dp[r-(1<<x)+1][x]);
}
void solve1(int l1,int r1,int l2,int r2){
	ll ans=LLONG_MIN;
	for(int i=l1;i<=r1;i++){
		ll minn=LLONG_MAX;
		for(int j=l2;j<=r2;j++) minn=min(minn,a[i]*b[j]);
		ans=max(ans,minn);
	}
	cout<<ans<<'\n';
}
void solve(int l1,int r1,int l2,int r2){
	ll maxb=query_max(dp3,l2,r2),minb=query_min(dp4,l2,r2),ans=LLONG_MIN,x;
	if(sum[r1]-sum[l1-1]>0) ans=max(ans,0LL);
	if(maxb>=0){
		x=query_max(dp5,l1,r1);
		if(x>-(1LL<<60)) ans=max(ans,x*maxb);
	}
	else{
		x=query_min(dp2,l1,r1);
		if(x<0) ans=max(ans,x*maxb);
	}
	if(minb>=0){
		x=query_max(dp1,l1,r1);
		if(x>0) ans=max(ans,x*minb);
	}
	else{
		x=query_min(dp6,l1,r1);
		if(x<(1LL<<60)) ans=max(ans,x*minb);
	}
	cout<<ans<<'\n';
	return;
}
int main(){
	ios::sync_with_stdio(false);cin.tie(0);cout.tie(0);
	cin>>n>>m>>q;
	for(int i=1;i<=n;i++) cin>>a[i];
	for(int i=1;i<=m;i++) cin>>b[i];
	init();
	bool flg=(n<=1000&&m<=1000&&q<=1000);
	int l1,r1,l2,r2;
	while(q--){
		cin>>l1>>r1>>l2>>r2;
		if(flg) solve1(l1,r1,l2,r2);
		else solve(l1,r1,l2,r2);
	}
	return 0;
}
```
