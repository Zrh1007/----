    long long cnt=0;

    }
    return cnt<=k;
}

        else l=mid+1;
    }
    cout<<ans;

## P1824 [USACO05FEB] 进击的奶牛 Aggressive Cows G

```cpp
#include<bits/stdc++.h>
using namespace std;
int x[100100000],n,c;
bool check(int mid){
    int cnt=1;
    int per=x[0];
    for(int i=1;i<n;i++){
        if(x[i]-per>=mid){
            cnt++;
            per=x[i];
        }
    }
    if(cnt>=c) return true;
    else return false;
}
int bsearch_2(int l,int r){
    while(l < r){
        int mid = l + r + 1 >> 1; // 找出中间值的下标
        if(check(mid)) l = mid; // 划分为[mid, r]
        else r = mid - 1; // 划分为[l, mid - 1]
    }
    return l;
}
int main(){
    cin>>n>>c;
    for(int i=0;i<n;i++) cin>>x[i];
    sort(x,x+n);
    int ans=bsearch_2(0,x[n-1]-x[0]);
    cout<<ans;
    return 0;
}
```
## P2678 [NOIP 2015 提高组] 跳石头
```cpp
#include<bits/stdc++.h>
using namespace std;
int l,n,m;
int d[50005];
int main(){
    cin>>l>>n>>m;
    for(int i=1;i<=n;i++) cin>>d[i];
    d[0]=0;
    d[n+1]=l;
    n+=2;
    int l=1,r=l;
    while(l<r){
        int mid=(l+r)>>1;
        int cnt=0;
        for(int i=1;i<n;i++){
            if(d[i+1]-d[i]<mid) cnt++;
        }
        if(cnt<=m) r=mid;
        else l=mid+1;
    }
    cout<<l;
    return 0;
}
```

## P9952 [USACO20OPEN] Social Distancing I B
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,ans;
string s;
int pos[100000];
int main(){
    cin>>n>>s;
    int last=0;
    for(int i=0;i<n;i++){
        if(s[i]=='1') pos[last++]=i;
    }
    for(int i=0;i<last-1;i++){
        int d=pos[i+1]-pos[i];
        ans=max(ans,d);
    }
    cout<<ans/2;
    return 0;
}
```
## P1083 [NOIP 2012 提高组] 借教室
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,r[1000005],d[1000005],s[1000005],t[1000005];
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++) cin>>r[i];
    for(int i=1;i<=m;i++) cin>>d[i]>>s[i]>>t[i];
    for(int i=1;i<=m;i++){
        bool flag=1;
        for(int j=s[i];j<=t[i];j++){
            if(r[j]<d[i]){
                flag=0;
                cout<<-1<<'\n'<<i<<'\n';
                return 0;
            }
            r[j]-=d[i];
        }
        if(!flag) break;
    }
    return 0;
}
```
