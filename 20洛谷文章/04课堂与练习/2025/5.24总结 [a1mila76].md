# 题目总结
## T1

### 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
int n;
unsigned long long s,l;
int a[200005],h[200005];
int main(){
    cin>>n>>s>>l;
    for(int i=1;i<=n;i++) cin>>h[i];
    for(int i=1;i<=n;i++) cin>>a[i];
    unsigned long long t=0;
    while(1){
        unsigned long long sum=0;
        int cnt=0;
        for(int i=1;i<=n;i++){
            if(h[i]+a[i]*t>=l){
                cnt++;
                sum+=h[i]+a[i]*t;
            }
        }
        if(sum>=s) break;
        t++;
    }
    cout<<t;
    return 0;
}
```

### 代码
```cpp
#include <bits/stdc++.h>
using namespace std;
int n,k,h[100005],w[100005];
int main(){
    cin>>n>>k;
    for(int i=1;i<=n;i++)cin>>h[i]>>w[i];
    int l=1,r=1e5,ans=0;
    while(l<=r){
        int mid=(l+r)/2;
        long long sum=0;
        for(int i=1;i<=n;i++){
            sum+=(long long)(h[i]/mid)*(w[i]/mid);
        }
        if(sum>=k){
            ans=mid;
            l=mid+1;
        }else{
            r=mid-1;
        }
    }
    cout<<ans;
    return 0;
}
```

## T4

### 代码
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m;
int a[100010];
int main(){
    cin>>n>>m;
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }
    int ans=1;
    int sum=0;
    for(int i=1;i<=n;i++){
        sum+=a[i];
        ans=max(ans,a[i]);
    }
    int l=ans,r=sum;
    while(l<r){
        int mid=(l+r)/2;
        int cnt=0;
        int s=0;
        for(int i=1;i<=n;i++){
            if(s+a[i]>mid){
                cnt++;
                s=a[i];
            }else{
                s+=a[i];
            }
        }
        if(s)cnt++;
        if(cnt>m)l=mid+1;
        else r=mid;
    }
    cout<<l;
    return 0;
}
```

## P4058 [code+#1]木材

#include<bits/stdc++.h>
using namespace std;
int n,s,l;
int h[200005],a[200005];
int main(){
    cin>>n>>s>>l;
    for(int i=0;i<n;i++) cin>>h[i];
    for(int i=0;i<n;i++) cin>>a[i];
    int l1=0,r1=1000000000;
    while(l1<r1){
        int mid=(l1+r1)/2;
        long long sum=0;
        for(int i=0;i<n;i++){
            if(h[i]+1LL*mid*a[i]>=l) sum+=h[i]+1LL*mid*a[i];
        }
        if(sum>=s) r1=mid;
        else l1=mid+1;
    }
    cout<<l1;
    return 0;
}

## P8647 [蓝桥杯 2017 省 AB] 分巧克力
```cpp
#include<iostream>
using namespace std;
int n,k,a[100010],b[100010];
int f(int x){
	int ans=0;
	for(int i=1;i<=n;i++) ans+=(a[i]/x)*(b[i]/x);
	return ans>=k;
}
int main(){
	cin>>n>>k;
	for(int i=1;i<=n;i++) cin>>a[i]>>b[i];
	int l=1,r=100000;
	while(l<r){
		int mid=(l+r+1)>>1;
		if(f(mid)) l=mid;
		else r=mid-1;
	}
	cout<<l;
    return 0;
}
```
