# B2064 斐波那契数列
## 思路：
- 比较简单，直接用循环来实现即可。
```cpp
#include<bits/stdc++.h>
using namespace std;
long long n,f[40];
/*
①确定状态
f[i]表示第i个斐波那契数
②确定答案
f[n]
③确定转移方程
f[i]=f[i-1]+f[i-2]
④确定边界条件
f[1]=1,f[2]=1
*/
int main(){
	cin>>n;
	f[1]=1;
	f[2]=1;
	for(int i=3;i<=35;i++) f[i]=f[i-1]+f[i-2];
	while(n--){
		int a;
		cin>>a;
		cout<<f[a]<<'\n';
	}
	return 0;
} 
```
# P1115 最大子段和
## 思路：
- 模板题，用动态规划来解决。
```cpp
#include<bits/stdc++.h>
using namespace std;
int f[200010],n,a[200010];
/*
①确定状态
f[i]表示以第i个元素结尾的最大子段和
②确定答案
max{f[1]~f[n]}
③确定状态转移方程
f[i]=max{f[i-1]+a[i],a[i]}
④处理边界条件和初始值
f[1]=a[1]
f[2]=max{a[1],a[2]}
*/
int main(){
	int n;    
	cin>>n;    
	for(int i=1;i<=n;i++){        
		cin>>a[i]; 
		f[i]=a[i];    
	}    
	int maxn=-0x3f3f3f3f;    
	for (int i=1;i<=n;i++){        
		f[i]=max(f[i-1]+a[i],a[i]); 
		maxn=max(maxn,f[i]);    
	}
	cout<<maxn;
	return 0;
}
```
# B3637 最长上升子序列
## 思路：
- 模板题
```cpp
#include<bits/stdc++.h>
using namespace std;
int a[5010],n,f[5010];
/*
①确定状态 
f[i]表示以第i个元素结尾的最长上升子序列的长度
②确定答案 
max{f[1]~f[n]}
③确定状态转移方程 
f[i]=max{f[j]+1,f[i]}
④处理边界条件和初始值
f[i]=1
*/
int main(){
	cin>>n;
	for(int i=1;i<=n;i++){
		cin>>a[i];
	}
	for(int i=1;i<=n;i++){
		f[i]=1;
		for(int j=1;j<i;j++){
			if(a[i]>a[j]){
				f[i]=max(f[i],f[j]+1);
			}
		}
	}
	int maxn=-1; 
	for(int i=1;i<=n;i++){
		maxn=max(maxn,f[i]);
	}
	cout<<maxn;
	return 0;
}
```
# P6702 [COCI2010-2011#7] ŠEĆER
## 思路：
- 模拟题，直接枚举即可。
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,minn=0x3f3f3f3f;
int main(){
	cin>>n;
	for(int i=0;i<=5000;i++){
		for(int j=0;j<=5000;j++){
			if(i*3+j*5==n){
				minn=min(minn,i+j);
			}
		}
	}
	if(minn==0x3f3f3f3f){
		cout<<-1;
		return 0;
	}
	cout<<minn;
	return 0;
}
```
# P1057 [NOIP2008 普及组] 传球游戏
## 思路：
- dp题，f[i][j]表示第i局，第j个人拿着球的情况下，有几种方案。
```cpp
#include<bits/stdc++.h>
using namespace std;
int n,m,f[40][40];
/*
①确定状态
f[i][j]表示第i局，第j个人拿着球的情况下，有几种方案。
②确定答案
f[m][1]
③确定状态转移方程
f[i][j]=f[i-1][n]+f[i-1][j+1];
f[i][j]=f[i-1][j-1]+f[i-1][1];
f[i][j]=f[i-1][j-1]+f[i-1][j+1];
④处理边界条件和初始值
f[0][1]=1;
*/
int main(){
	cin>>n>>m;
	f[0][1]=1;
	for(int i=1;i<=m;i++){
		for(int j=1;j<=n;j++){
			if(j==1){
				f[i][j]=f[i-1][n]+f[i-1][j+1];
			}
			else if(j==n){
				f[i][j]=f[i-1][j-1]+f[i-1][1];
			}
			else{
				f[i][j]=f[i-1][j-1]+f[i-1][j+1];
			}
		}
	}
	cout<<f[m][1];
	return 0;
}
```
