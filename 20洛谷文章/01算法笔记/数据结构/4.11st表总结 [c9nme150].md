# ST表


> [!info]- 定义
>
> ST 表（Sparse Table，稀疏表）是用于解决一个区间内反复查询贡献问题的数据结构。
> ST 表的核心思想是预处理一个数组，使得在 O(1) 的时间内能够查询任意区间的最小值、最大值或其他可结合的函数结果。

# 例题

## P3865 【模板】ST 表 & RMQ 问题
```cpp
#include<bits/stdc++.h>
using namespace std;
int dp[100010][30];
int n,m,a[100010];
void init(){
    memset(dp,0,sizeof(dp));
    for(int i=1;i<=n;i++) dp[i][0]=a[i];
    for(int j=1;j<=log2(n);j++){
        for(int i=1;i+(1<<j)-1<=n;i++){
            dp[i][j]=max(dp[i][j-1],dp[i+(1<<(j-1))][j-1]);
        }
    }
}
int main(){
    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);
    //freopen(".in","r",stdin);
    //freopen(".out","w",stdout);
    cin>>n>>m;
    for(int i=1;i<=n;i++) cin>>a[i];
    init();
    while(m--){
        int l,r;
        cin>>l>>r;
        int k=log2(r-l+1);
        cout<<max(dp[l][k],dp[r-(1<<k)+1][k])<<'\n';
    }
    return 0;
}
```

## P2251 质量检测

```cpp
#include<bits/stdc++.h>

int n,m,a;
int x[100010];
deque<int> q;

    ios::sync_with_stdio(0),cin.tie(0),cout.tie(0);

    for(int i=1;i<=n;i++){
        cin>>x[i];
    }
    for(int i=1;i<=n;i++){
        while(q.size()&&x[q.back()]>=x[i]) q.pop_back();
        q.push_back(i);
        while(q.size()&&q.front()<=i-m) q.pop_front();
        if(i>=m) cout<<x[q.front()]<<'\n';
    }
    return 0;
}
```
